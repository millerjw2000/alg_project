** BNF Grammar used for this project **
program             ::= statement* EOF

statement           ::= assignment
                   |    if_statement

assignment          ::= VAR ASSIGN expression SEMICOL

if_statement        ::= IF expression THEN statement*
                        (ELSE statement*)?
                        END SEMICOL

expression          ::= logical_or

logical_or          ::= logical_and (OR logical_and)*

logical_and         ::= comparison (AND comparison)*

comparison          ::= additive (comparison_op additive)?

comparison_op       ::= GT | LT | GTE | LTE | EQ | NEQ

additive            ::= multiplicative ((PLUS | MINUS) multiplicative)*

multiplicative      ::= unary ((MULT | DIV) unary)*

unary               ::= (PLUS | MINUS | NOT) unary
                        | primary

primary             ::= NUM
                        | VAR
                        | function_call
                        | LPAR expression RPAR

function_call       ::= VAR LPAR arguments? RPAR

arguments           ::= expression (COM expression)*
